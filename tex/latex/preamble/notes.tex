\def\@lecture{}%
\newcommand{\lecture}[3]{
    \ifthenelse{\isempty{#3}}{%
        \def\@lecture{#1.}%
    }{%
        \def\@lecture{#1. #3}%
    }%
    \section*{\@lecture}
    \addcontentsline{toc}{section}{\@lecture}
    \marginpar{\small\textsf{\mbox{#2}}}
    \medskip
}
